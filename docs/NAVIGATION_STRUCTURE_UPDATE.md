    };

    // Immediate processing for critical events

    // Buffer for batch processing

        type: 'dashboard_update',
        dashboardType,
        data: optimizedData,
        timestamp: new Date().toISOString()

    });

      // Load into analytics database
      const loadResult = await this.loadAnalyticsData(transformedData);
      
      // Update cache with fresh data
      await this.updateAnalyticsCache(tenantId, dataType, transformedData);
      
      syncJob.status = 'completed';
      syncJob.recordsProcessed = loadResult.recordCount;
      
    } catch (error) {
      syncJob.status = 'failed';
      syncJob.error = error.message;
      throw error;
    } finally {
      syncJob.endTime = Date.now();
      syncJob.duration = syncJob.endTime - syncJob.startTime;
      await this.logSyncJob(syncJob);
    }

    return syncJob;
  }

      this.buildPartitionQuery(partition, timeRange, aggregation)
